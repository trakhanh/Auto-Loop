### Task 1: Lõi — đọc thời gian và tính danh sách đoạn

**Files:**
- Create: `loopstation/core/cutter.py`
- Test: `tests/test_cutter.py`

**Interfaces:**
- Produces:
  - `parse_time(text: str) -> float` — `hh:mm:ss[.f]`, `mm:ss[.f]`, `ss[.f]`; khi có giờ thì phút/giây < 60; khi `mm:ss` thì giây < 60; sai → `ValueError(f"Thời gian «{text}» sai")`.
  - `parse_marks(text: str) -> list[float]` — tách bằng `,` `;` xuống dòng; ô trống → `[]`.
  - `parse_ranges(text: str) -> list[tuple[float, float]]` — mỗi mục `A-B` (chấp nhận `-` `–` `—`, có/không dấu cách); A ≥ B → `ValueError("Khoảng «…» sai: đầu phải trước cuối")`.
  - `fmt_time(sec: float) -> str` → `"00:01:15"` (làm tròn xuống giây; ≥ 1 giờ vẫn `hh:mm:ss`).
  - `@dataclass Segment(start: float, dur: float)`.
  - `@dataclass SplitSettings(mode="fixed", parts=2, fixed=30.0, rmin=10.0, rmax=45.0, marks="", ranges="", exact=True, aspect="keep", fit="blur", per_video=True, open_after=False)`; `SPLIT_MODES = ("parts", "fixed", "random", "marks", "ranges")`.
  - `plan_segments(duration: float, s: SplitSettings, rng: random.Random | None = None) -> tuple[list[Segment], str]` — trả (đoạn, ghi chú vàng hoặc ""); cài đặt sai → `ValueError` câu tiếng Việt.

import random, pytest
from loopstation.core import cutter as c

def test_parse_time_formats():
    assert c.parse_time("00:01:15") == 75
    assert c.parse_time("1:15") == 75
    assert c.parse_time("75") == 75
    assert c.parse_time(" 00:01:15.5 ") == 75.5
    assert c.parse_time("1:00:00") == 3600

@pytest.mark.parametrize("bad", ["", "abc", "00:75:00", "1:2:3:4", "-5", "1:60"])
def test_parse_time_bad(bad):
    with pytest.raises(ValueError):
        c.parse_time(bad)

def test_parse_marks_and_ranges():
    assert c.parse_marks("00:01:15, 00:05:30\n12:00;") == [75, 330, 720]
    assert c.parse_marks("  ") == []
    assert c.parse_ranges("00:01:10-00:01:45, 5:00 – 5:30") == [(70, 105), (300, 330)]
    with pytest.raises(ValueError, match="00:02:00-00:01:00"):
        c.parse_ranges("00:02:00-00:01:00")
    with pytest.raises(ValueError, match="00:75:00"):
        c.parse_marks("00:01:00, 00:75:00")

def test_fmt_time():
    assert c.fmt_time(75.9) == "00:01:15" and c.fmt_time(3725) == "01:02:05"

def segs(d, **kw):
    out, note = c.plan_segments(d, c.SplitSettings(**kw), random.Random(1))
    return [(round(x.start, 3), round(x.dur, 3)) for x in out], note

def test_parts():
    assert segs(90, mode="parts", parts=3)[0] == [(0, 30), (30, 30), (60, 30)]
    s, note = segs(2.5, mode="parts", parts=10)          # mỗi phần < 1 s → hạ N
    assert len(s) == 2 and note

def test_fixed_tail():
    assert segs(95, mode="fixed", fixed=30)[0][-1] == (90, 5)      # đoạn cuối dư giữ lại
    assert len(segs(90.5, mode="fixed", fixed=30)[0]) == 3          # dư < 1 s → bỏ

def test_random_bounds_and_cover():
    s, _ = segs(600, mode="random", rmin=10, rmax=45)
    assert s[0][0] == 0 and abs(sum(d for _, d in s) - 600) < 1e-6
    assert all(10 - 1e-6 <= d <= 45 + 1e-6 for _, d in s[:-1])
    with pytest.raises(ValueError):
        segs(600, mode="random", rmin=50, rmax=10)

def test_random_short_video_one_piece():
    assert segs(8, mode="random", rmin=10, rmax=45)[0] == [(0, 8)]

def test_marks():
    s, note = segs(600, mode="marks", marks="00:05:30, 00:01:15, 00:01:15, 00:20:00")
    assert s == [(0, 75), (75, 255), (330, 270)] and "00:20:00" in note

def test_ranges():
    s, note = segs(100, mode="ranges", ranges="00:00:10-00:00:20, 00:01:30-00:02:00, 00:03:00-00:03:10")
    assert s == [(10, 10), (90, 10)] and "00:03:00" in note          # vượt cuối → cắt tới hết; ngoài video → bỏ

def test_empty_marks_is_error():
    with pytest.raises(ValueError):
        segs(100, mode="marks", marks="")

Quy tắc theo spec mục 3: `fixed` giữ đoạn dư ≥ 1 s; `parts` hạ N xuống `floor(duration)` khi `duration / N < 1` (ghi chú "Video dài … — chia tối đa N phần"); `marks` sắp xếp, bỏ trùng, bỏ mốc ≤ 0 hoặc ≥ thời lượng (ghi chú liệt kê mốc bỏ bằng `fmt_time`), ô trống → lỗi "Chưa nhập mốc nào"; `ranges` theo thứ tự gõ, đầu ≥ thời lượng → bỏ + ghi chú, cuối vượt → cắt tới hết, ô trống → lỗi; `random` thuật toán:

```python
t, out = 0.0, []
while duration - t > 1e-6:
    rest = duration - t
    if rest <= s.rmax and (rest <= s.rmin or not out):   # phần còn lại vừa một đoạn
        d = rest
    else:
        d = rng.uniform(s.rmin, s.rmax)
        tail = rest - d
        if tail < s.rmin:                                  # đoạn cuối quá ngắn
            if rest <= s.rmax: d = rest                    # nhập vào đoạn này
            elif tail < 1: d = rest                        # (không xảy ra khi rmin ≥ 1)
    out.append(Segment(t, d)); t += d
```
`rmin < 1` hoặc `rmin > rmax` → `ValueError`. `rng=None` → `random.Random()`.
